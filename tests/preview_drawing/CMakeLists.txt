cmake_minimum_required(VERSION 3.16)
project(PreviewDrawingSchedulerTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Core)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Core)
enable_testing()
add_executable(preview_drawing_tests scheduler_test.cpp)
target_include_directories(preview_drawing_tests PRIVATE ../../toonz/sources/toonz)
target_link_libraries(preview_drawing_tests PRIVATE Qt${QT_VERSION_MAJOR}::Core)
# Keep assertions active in Release builds too.
if(MSVC)
  target_compile_options(preview_drawing_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(preview_drawing_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME preview_drawing_scheduler COMMAND preview_drawing_tests)
