cmake_minimum_required(VERSION 3.10)
project(render_scale_test LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../toonz/sources")
add_executable(render_scale_test render_scale_test.cpp
               "${SOURCE_ROOT}/common/tgeometry/tgeometry.cpp")
target_include_directories(render_scale_test PRIVATE "${SOURCE_ROOT}/include")
target_compile_definitions(render_scale_test PRIVATE TGEOMETRY_EXPORTS)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_compile_definitions(render_scale_test PRIVATE LINUX)
elseif(APPLE)
  target_compile_definitions(render_scale_test PRIVATE MACOSX)
elseif(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
  target_compile_definitions(render_scale_test PRIVATE FREEBSD)
endif()
enable_testing()
add_test(NAME render_scale COMMAND render_scale_test)
