cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(OpenToonzBulletExploration LANGUAGES CXX)
    set(OT_BULLET_STANDALONE ON)
    enable_testing()
endif()

set(OT_BULLET_SOURCE_DIR "" CACHE PATH "Optional Bullet SDK source checkout (tested with 3.25)")
option(BUILD_BULLET_TESTS "Build Bullet exploration regression tests" ${OT_BULLET_STANDALONE})

# No download during configuration. A supplied SDK builds only the three CPU
# components; Bullet's top-level project, examples and global flags are unused.
function(ot_bullet_dependencies)
    if(OT_BULLET_SOURCE_DIR)
        get_filename_component(BULLET_PHYSICS_SOURCE_DIR "${OT_BULLET_SOURCE_DIR}" ABSOLUTE)
        file(STRINGS "${BULLET_PHYSICS_SOURCE_DIR}/VERSION" BULLET_VERSION LIMIT_COUNT 1)
        set(BUILD_SHARED_LIBS OFF)
        set(INSTALL_LIBS OFF)
        foreach(component LinearMath BulletCollision BulletDynamics)
            add_subdirectory("${BULLET_PHYSICS_SOURCE_DIR}/src/${component}" "sdk/${component}" EXCLUDE_FROM_ALL)
            set_target_properties(${component} PROPERTIES POSITION_INDEPENDENT_CODE ON)
        endforeach()
        set(OT_BULLET_LIBRARIES BulletDynamics BulletCollision LinearMath PARENT_SCOPE)
        set(OT_BULLET_INCLUDES "${BULLET_PHYSICS_SOURCE_DIR}/src" PARENT_SCOPE)
    else()
        find_package(Bullet REQUIRED)
        set(OT_BULLET_LIBRARIES ${BULLET_DYNAMICS_LIBRARY} ${BULLET_COLLISION_LIBRARY} ${BULLET_MATH_LIBRARY} PARENT_SCOPE)
        set(OT_BULLET_INCLUDES ${BULLET_INCLUDE_DIRS} PARENT_SCOPE)
    endif()
endfunction()
ot_bullet_dependencies()

add_library(ot_bullet STATIC bulletworld.cpp)
set_target_properties(ot_bullet PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(ot_bullet PUBLIC cxx_std_17)
target_include_directories(ot_bullet PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(ot_bullet SYSTEM PRIVATE ${OT_BULLET_INCLUDES})
target_link_libraries(ot_bullet PUBLIC ${OT_BULLET_LIBRARIES})

if(NOT OT_BULLET_STANDALONE)
    add_library(ot_bullet_stage STATIC bulletstage.cpp)
    set_target_properties(ot_bullet_stage PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_include_directories(ot_bullet_stage PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_link_libraries(ot_bullet_stage PUBLIC ot_bullet toonzlib tnzbase tnzcore)
endif()

    endif()
