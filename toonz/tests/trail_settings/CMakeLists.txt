cmake_minimum_required(VERSION 3.16)
project(TrailSettingsTests LANGUAGES CXX)

if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Must match tnzcore" FORCE)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
get_filename_component(TOONZ_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/../../sources" ABSOLUTE)

add_executable(trailcyclestate_test trailcyclestate_test.cpp)
target_include_directories(trailcyclestate_test PRIVATE "${TOONZ_SOURCES}" "${TOONZ_SOURCES}/include")
if(MSVC)
  target_compile_options(trailcyclestate_test PRIVATE /W4 /WX)
else()
  target_compile_options(trailcyclestate_test PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME trailcyclestate COMMAND trailcyclestate_test)

add_executable(trailstamp_test trailstamp_test.cpp)
target_include_directories(trailstamp_test PRIVATE "${TOONZ_SOURCES}")
target_include_directories(trailstamp_test SYSTEM PRIVATE "${TOONZ_SOURCES}/include")
if(MSVC)
  target_compile_options(trailstamp_test PRIVATE /W4 /WX)
else()
  target_compile_options(trailstamp_test PRIVATE -Wall -Wextra -Werror)
endif()
if(APPLE)
  target_compile_definitions(trailstamp_test PRIVATE MACOSX)
elseif(UNIX)
  target_compile_definitions(trailstamp_test PRIVATE LINUX)
endif()
add_test(NAME trailstamp COMMAND trailstamp_test)

set(TNZCORE_LIBRARY "" CACHE FILEPATH "tnzcore built from this checkout and build configuration")
if(TNZCORE_LIBRARY)
  find_package(Qt5 REQUIRED COMPONENTS Core Gui OpenGL Multimedia Network)
  include(TestBigEndian)
  test_big_endian(IS_BIG_ENDIAN)
  set(PLI_SOURCES "${TOONZ_SOURCES}/image/pli/pli_io.cpp"
      "${TOONZ_SOURCES}/image/pli/tiio_pli.cpp" "${TOONZ_SOURCES}/image/pli/tags.cpp")
  foreach(name IN ITEMS pli trailstyle)
    add_executable(${name}_test ${name}_test.cpp ${PLI_SOURCES})
    target_include_directories(${name}_test SYSTEM PRIVATE "${TOONZ_SOURCES}/include")
    target_include_directories(${name}_test PRIVATE "${TOONZ_SOURCES}")
    target_link_libraries(${name}_test PRIVATE "${TNZCORE_LIBRARY}" Qt5::Core Qt5::Gui)
    if(IS_BIG_ENDIAN)
      target_compile_definitions(${name}_test PRIVATE TNZ_LITTLE_ENDIAN=0)
    else()
      target_compile_definitions(${name}_test PRIVATE TNZ_LITTLE_ENDIAN=1)
    endif()
    if(APPLE)
      target_compile_definitions(${name}_test PRIVATE MACOSX i386 GL_SILENCE_DEPRECATION)
      add_custom_command(TARGET ${name}_test POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E create_symlink "${TNZCORE_LIBRARY}"
                "$<TARGET_FILE_DIR:${name}_test>/libtnzcore.dylib")
    elseif(UNIX)
      target_compile_definitions(${name}_test PRIVATE LINUX)
    endif()
    add_test(NAME ${name} COMMAND ${name}_test)
  endforeach()
endif()
